# === Peta Jalur Distribusi Logistik (PyDeck) ===
MAP_ROUTES = [
    # label, src_lon, src_lat, tgt_lon, tgt_lat, hex_color, width
    ("IMIP - Morowali",      122.15, -2.82, 113.8, 22.8,  [230, 25,  25],  6),
    ("GNI - Morowali Utara", 121.32, -1.91, 113.8, 22.8,  [255, 140, 0],   5),
    ("VDNI/OSS - Konawe",    122.42, -3.83, 113.8, 22.8,  [0,  112, 220],  5),
    ("ANTAM - Kolaka",       121.60, -4.18, 135.0, 35.0,  [0,  180, 80],   4),
    ("PT Vale - Sorowako",   121.34, -2.56, 135.0, 35.0,  [180, 0,  200],  4),
]

# Build dataframes
_routes, _dots = [], []
for name, slon, slat, tlon, tlat, col, w in MAP_ROUTES:
    _routes.append({"label": name, "src_lon": slon, "src_lat": slat,
                    "tgt_lon": tlon, "tgt_lat": tlat, "color": col, "width": w})
    # smelter dot (radius 15km)
    _dots.append({"label": name, "lon": slon, "lat": slat, "color": col, "r": 15000})
    
# destination dots (radius 15km) - only 2 nodes as requested
_dots.append({"label": "China (Pasar Utama)", "lon": 113.8, "lat": 22.8, "color": [97, 97, 97], "r": 20000})
_dots.append({"label": "Jepang/Korea", "lon": 135.0, "lat": 35.0, "color": [97, 97, 97], "r": 20000})

# Text labels with large geographic offsets so they don't overlap
_text_labels = [
    ("IMIP\n(Morowali)",      122.15+1.2, -2.82-0.5),
    ("GNI\n(Morut)",          121.32-1.5, -1.91-0.2),
    ("VDNI/OSS\n(Konawe)",    122.42+1.5, -3.83+0.3),
    ("ANTAM\n(Kolaka)",       121.60-1.8, -4.18+0.7),
    ("PT Vale\n(Sorowako)",   121.34-1.5, -2.56+0.8),
    ("China\n(Pasar Utama)",  113.8, 24.8),
    ("Jepang /\nKorea",       135.0, 37.0),
]
_text_rows = [{"txt": t, "lon": lo, "lat": la} for t, lo, la in _text_labels]

df_routes = pd.DataFrame(_routes)
df_dots   = pd.DataFrame(_dots)
df_text   = pd.DataFrame(_text_rows)

arc_layer = pdk.Layer(
    "ArcLayer",
    data=df_routes,
    get_source_position=["src_lon", "src_lat"],
    get_target_position=["tgt_lon", "tgt_lat"],
    get_source_color="color",
    get_target_color="color",
    get_width="width",
    pickable=True,
    auto_highlight=True,
)

dot_layer = pdk.Layer(
    "ScatterplotLayer",
    data=df_dots,
    get_position=["lon", "lat"],
    get_fill_color="color",
    get_line_color=[255, 255, 255],
    stroked=True,
    line_width_min_pixels=2,
    get_radius="r",
    radius_min_pixels=4,
    radius_max_pixels=15,
    pickable=True,
)

text_layer = pdk.Layer(
    "TextLayer",
    data=df_text,
    get_position=["lon", "lat"],
    get_text="txt",
    get_size=13,
    get_color=[20, 20, 20],
    get_background_color=[255, 255, 255, 210],
    background_padding=[5, 3, 5, 3],
    get_text_anchor="'middle'",
    get_alignment_baseline="'center'",
    pickable=False,
)

view_state = pdk.ViewState(
    longitude=122.0,
    latitude=12.0,
    zoom=3.0,
    pitch=0,
    bearing=0,
)

r = pdk.Deck(
    layers=[arc_layer, dot_layer, text_layer],
    initial_view_state=view_state,
    map_style="light",
    tooltip={"html": "<b>{label}</b>"},
)

st.markdown("#### 1.5 Peta Jalur Distribusi Logistik Nikel Sulawesi")
st.pydeck_chart(r, use_container_width=True, height=500)
